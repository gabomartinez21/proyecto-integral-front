/** Solo deja pasar a usuarios con sesiÃ³n; si no, redirige al login. */
export function RequireAuth({ children }: { children: ReactNode }) {
  const user = useCurrentUser()
  const location = useLocation()
  if (!user) return <Navigate to={ROUTES.login} replace state={{ from: location.pathname }} />
  return children
}

/** Evita mostrar el login a quien ya iniciÃ³ sesiÃ³n. */
export function GuestOnly({ children }: { children: ReactNode }) {
  const user = useCurrentUser()
  if (user) return <Navigate to={ROUTES.dashboard} replace />
  return children
}

import { useEffect, useRef, type ReactNode } from 'react'

interface ModalProps {
  open: boolean
  title: string
  onClose: () => void
  children: ReactNode
}

/** Modal accesible basado en `<dialog>` nativo (foco atrapado y cierre con Esc). */
export function Modal({ open, title, onClose, children }: ModalProps) {
  const ref = useRef<HTMLDialogElement>(null)

  useEffect(() => {
    const dialog = ref.current
    if (!dialog) return
    if (open && !dialog.open) dialog.showModal()
    if (!open && dialog.open) dialog.close()
  }, [open])

  return (
    <dialog
      ref={ref}
      onClose={onClose}
      aria-labelledby="modal-title"
      className="m-auto max-h-[90vh] w-[min(650px,94vw)] rounded-[17px] border border-line p-[27px] text-ink shadow-[0_30px_100px_#16372233] print:hidden"
    >
      {open && (
        <>
          <div className="flex items-center justify-between gap-[15px]">
            <h2 id="modal-title">{title}</h2>
            <button
              type="button"
              onClick={onClose}
              aria-label="Cerrar"
              className="rounded-lg p-1.5 text-[22px] leading-none hover:bg-canvas"
            >
              Ã—
            </button>
          </div>
          {children}
        </>
      )}
    </dialog>
  )
}

export const changeOrderStatus: CommandHandler<'order/changeStatus'> = (draft, p) => {
  const order = draft.orders.find((x) => x.id === p.id)
  if (!order) fail('Pedido inexistente.')
  if (!ORDER_TRANSITIONS[order.status]?.includes(p.status)) fail('Cambio de estado no permitido.')
  const needs = orderNeeds(order)

  if (p.status === 'Confirmado') {
    const reserved = reservedQuantities(draft, order.id)
    for (const [ingredientId, quantity] of Object.entries(needs)) {
      const ingredient = draft.ingredients.find((i) => i.id === ingredientId)
      if (!ingredient || round(ingredient.stock - (reserved[ingredientId] ?? 0)) < quantity)
        fail(`Stock insuficiente: ${ingredient?.name ?? 'insumo'}. Revisa la lista de compras.`)
    }
  }

  if (p.status === 'Entregado') {
    for (const [ingredientId, quantity] of Object.entries(needs)) {
      const ingredient = draft.ingredients.find((i) => i.id === ingredientId)
      if (!ingredient || ingredient.stock < quantity) fail('Stock insuficiente.')
      ingredient.stock = round(ingredient.stock - quantity)
      draft.movements.push({
        id: createId(),
        date: new Date().toISOString(),
        ingredient: ingredient.name,
        quantity: -quantity,
        unit: ingredient.unit,
        note: `Entrega de pedido #${order.number}`,
      })
    }
  }

  order.status = p.status as typeof order.status
}

import { ACTIVE_ORDER_STATUSES } from './constants'
import type { Ingredient, KitchenState, Order, QuantityMap, ShoppingItem } from './types'
import { addQuantity, round } from './utils'

/** Ingredientes totales que consume un pedido. */
export function orderNeeds(order: Order): QuantityMap {
  const result: QuantityMap = {}
  for (const line of order.items)
    for (const r of line.recipe) addQuantity(result, r.ingredientId, r.quantity * line.quantity)
  return result
}

function sumNeeds(orders: Order[]): QuantityMap {
  const result: QuantityMap = {}
  for (const o of orders)
    for (const [k, v] of Object.entries(orderNeeds(o))) addQuantity(result, k, v)
  return result
}

/** Cantidades apartadas por pedidos confirmados (opcionalmente excluyendo uno). */
export function reservedQuantities(state: KitchenState, exceptOrderId?: string): QuantityMap {
  return sumNeeds(state.orders.filter((o) => o.status === 'Confirmado' && o.id !== exceptOrderId))
}

export function availableStock(ingredient: Ingredient, reserved: QuantityMap): number {
  return round(ingredient.stock - (reserved[ingredient.id] ?? 0))
}

export function isLowStock(ingredient: Ingredient, reserved: QuantityMap): boolean {
  return availableStock(ingredient, reserved) <= ingredient.minimum
}

export function lowStockIngredients(state: KitchenState): Ingredient[] {
  const reserved = reservedQuantities(state)
  return state.ingredients.filter((i) => isLowStock(i, reserved))
}

/**
 * Lista de compras para los pedidos activos entre `from` y `to` (inclusive).
 * El stock disponible descuenta lo reservado por pedidos confirmados fuera del rango.
 */
export function shoppingList(state: KitchenState, from = '', to = '9999'): ShoppingItem[] {
  const inRange = (o: Order) => o.date >= from && o.date <= to
  const required = sumNeeds(
    state.orders.filter(
      (o) => (ACTIVE_ORDER_STATUSES as readonly string[]).includes(o.status) && inRange(o),
    ),
  )
  const reservedOutside = sumNeeds(
    state.orders.filter((o) => o.status === 'Confirmado' && !inRange(o)),
  )

  return state.ingredients
    .filter((i) => required[i.id])
    .map((i) => {
      const needed = required[i.id] ?? 0
      const available = round(Math.max(0, i.stock - (reservedOutside[i.id] ?? 0)))
      return { ...i, required: needed, available, buy: round(Math.max(0, needed - available)) }
    })
}

export function createId(): string {
  if (typeof crypto.randomUUID === 'function') return crypto.randomUUID()
  return Array.from(crypto.getRandomValues(new Uint8Array(16)), (x) =>
    x.toString(16).padStart(2, '0'),
  ).join('')
}

  map[key] = round((map[key] ?? 0) + value)

/**
 * Ejecuta comandos de dominio desde la UI: expone el error para mostrarlo en
 * el formulario y avisa al usuario cuando se guarda correctamente.
 */
export function useCommand() {
  const dispatch = useKitchenStore((s) => s.dispatch)
  const pending = useKitchenStore((s) => s.busy)
  const [error, setError] = useState<string | null>(null)

  const run = useCallback(
    async (command: KitchenCommand, successMessage = 'Guardado correctamente') => {
      setError(null)
      try {
        await dispatch(command)
        toast(successMessage)
        return true
      } catch (e) {
        setError(errorMessage(e))
        return false
      }
    },
    [dispatch],
  )

  return { run, error, pending, clearError: () => setError(null) }
}

interface KitchenStore {
  status: LoadStatus
  loadError: string | null
  data: KitchenState
  revision: number
  busy: boolean
  load: () => Promise<void>
  /** Ejecuta un comando de dominio y lo persiste. Lanza un error si falla. */
  dispatch: (command: KitchenCommand) => Promise<void>
}

export const useKitchenStore = create<KitchenStore>()((set, get) => ({
  status: 'idle',
  loadError: null,
  data: createInitialState(),
  revision: 0,
  busy: false,

  load: async () => {
    if (get().status === 'loading') return
    set({ status: 'loading', loadError: null })
    try {
      const { data, revision } = await kitchenRepository.read()
      set({ data, revision, status: 'ready' })
    } catch (error) {
      set({ status: 'error', loadError: errorMessage(error) })
    }
  },

  dispatch: async (command) => {
    const user = useAuthStore.getState().user
    if (!user) throw new Error('Inicia sesiÃ³n.')
    if (get().busy) throw new Error('Espera a que termine la operaciÃ³n anterior.')

    set({ busy: true })
    try {
      // Control de concurrencia optimista: si otra pestaÃ±a guardÃ³ antes, no se sobrescribe.
      const saved = await kitchenRepository.read()
      if (saved.revision !== get().revision) {
        set({ data: saved.data, revision: saved.revision })
        throw new Error('Los datos cambiaron en otra pestaÃ±a. Revisa y vuelve a guardar.')
      }
      const next = {
        data: applyCommand(saved.data, command, user.role),
        revision: saved.revision + 1,
      }
      await kitchenRepository.write(next)
      set(next)
    } finally {
      set({ busy: false })
    }
  },
}))
